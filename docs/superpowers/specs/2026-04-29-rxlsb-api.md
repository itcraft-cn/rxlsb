```rust
impl CellData {

    }

        matches!(self, CellData::Number(_))
    }
    
    pub fn is_date(&self) -> bool {
        matches!(self, CellData::Date(_))
    }
    
    pub fn is_bool(&self) -> bool {
        matches!(self, CellData::Bool(_))
    }
    
    pub fn is_blank(&self) -> bool {
        matches!(self, CellData::Blank)
    }
    
    pub fn is_error(&self) -> bool {
        matches!(self, CellData::Error(_))
    }
    
    pub fn as_text(&self) -> Option<&str> {
        match self {
            CellData::Text(s) => Some(s.as_str()),
            _ => None,
        }
    }
    
    pub fn as_number(&self) -> Option<f64> {
        match self {
            CellData::Number(n) => Some(*n),
            CellData::Date(d) => Some(excel_date_serial(d)),
            _ => None,
        }
    }
}
```

## 四、XlsbWriter API

### 4.1 Builder

```rust
pub struct XlsbWriterBuilder {
    path: Option<PathBuf>,
    template: Option<PathBuf>,
}

impl XlsbWriterBuilder {
    pub fn new() -> Self {
        Self {
            path: None,
            template: None,
        }
    }
    
    pub fn path(mut self, p: impl Into<PathBuf>) -> Self {
        self.path = Some(p.into());
        self
    }
    
    pub fn template(mut self, t: impl Into<PathBuf>) -> Self {
        self.template = Some(t.into());
        self
    }
    
    pub fn build(self) -> Result<XlsbWriter> {
        let path = self.path.ok_or(XlsbError::InvalidSheetIndex(0))?;
        
        if let Some(template) = self.template {
            XlsbWriter::from_template(&template, &path)
        } else {
            XlsbWriter::new(&path)
        }
    }
}
```

### 4.2 XlsbWriter结构

```rust
pub struct XlsbWriter {
    path: PathBuf,
    container: XlsbContainerWriter,
    sst: SstTable,
    styles: StylesRegistry,
    workbook: WorkbookWriter,
    sheets: Vec<SheetWriter>,
    current_sheet: Option<SheetWriter>,
    template_mode: bool,
}

impl XlsbWriter {
    pub fn builder() -> XlsbWriterBuilder {
        XlsbWriterBuilder::new()
    }
    
    fn new(path: &Path) -> Result<Self> {
        let container = XlsbContainerWriter::create(path)?;
        let sst = SstTable::new();
        let styles = StylesRegistry::new();
        let workbook = WorkbookWriter::new();
        
        Ok(Self {
            path: path.to_path_buf(),
            container,
            sst,
            styles,
            workbook,
            sheets: vec![],
            current_sheet: None,
            template_mode: false,
        })
    }
    
    fn from_template(template: &Path, output: &Path) -> Result<Self> {
        let reader = XlsbReader::builder().path(template).build()?;
        let sst = reader.sst.clone();
        let styles = reader.styles.clone();
        reader.close()?;
        
        let container = XlsbContainerWriter::create(output)?;
        let workbook = WorkbookWriter::new();
        
        Ok(Self {
            path: output.to_path_buf(),
            container,
            sst,
            styles,
            workbook,
            sheets: vec![],
            current_sheet: None,
            template_mode: true,
        })
    }
}
```

### 4.3 批量写入

```rust
impl XlsbWriter {
    pub fn write_batch(&mut self, sheet_name: &str,
                       supplier: impl CellSupplier,
                       row_count: usize, col_count: usize) -> Result<()> {
        let sheet_writer = SheetWriter::new(&mut self.sst, &mut self.styles)?;
        sheet_writer.write_batch(supplier, row_count, col_count)?;
        
        self.workbook.add_sheet(sheet_name);
        self.sheets.push(sheet_writer);
        Ok(())
    }
}
```

### 4.4 流式写入

```rust
impl XlsbWriter {
    pub fn start_sheet(&mut self, name: &str, col_count: usize) -> Result<()> {
        if self.current_sheet.is_some() {
            return Err(XlsbError::SheetNotStarted);
        }
        
        self.current_sheet = Some(SheetWriter::new_streaming(&mut self.sst, col_count)?);
        self.workbook.add_sheet(name);
        Ok(())
    }
    
    pub fn write_rows(&mut self, start_row: usize,
                      supplier: impl CellSupplier, row_count: usize) -> Result<()> {
        let sheet = self.current_sheet.as_mut()
            .ok_or(XlsbError::SheetNotStarted)?;
        sheet.write_rows(start_row, supplier, row_count)?;
        Ok(())
    }
    
    pub fn end_sheet(&mut self) -> Result<()> {
        let sheet = self.current_sheet.take()
            .ok_or(XlsbError::SheetNotStarted)?;
        self.sheets.push(sheet);
        Ok(())
    }
}
```

### 4.5 关闭与序列化

```rust
impl XlsbWriter {
    pub fn close(&mut self) -> Result<()> {
        self.serialize()?;
        self.container.finish()?;
        Ok(())
    }
    
    fn serialize(&mut self) -> Result<()> {
        self.container.add_entry("[Content_Types].xml",
            &XmlGen::content_types(self.sheets.len(), self.sst.count() > 0))?;
        self.container.add_entry("_rels/.rels", &RelsGen::root_rels())?;
        self.container.add_entry("docProps/app.xml",
            &XmlGen::app_xml(self.sheets.len()))?;
        self.container.add_entry("docProps/core.xml", &XmlGen::core_xml())?;
        self.container.add_entry("xl/theme/theme1.xml", &XmlGen::theme_xml())?;
        
        let workbook_data = self.workbook.serialize()?;
        self.container.add_entry("xl/workbook.bin", &workbook_data)?;
        
        self.container.add_entry("xl/_rels/workbook.bin.rels",
            &RelsGen::workbook_rels(self.sheets.len(), self.sst.count() > 0))?;
        
        let styles_data = self.styles.serialize()?;
        self.container.add_entry("xl/styles.bin", &styles_data)?;
        
        if self.sst.count() > 0 {
            let sst_data = self.sst.serialize()?;
            self.container.add_entry("xl/sharedStrings.bin", &sst_data)?;
        }
        
        for (i, sheet) in self.sheets.iter().enumerate() {
            let sheet_data = sheet.serialize()?;
            self.container.add_entry(&format!("xl/worksheets/sheet{}.bin", i + 1), &sheet_data)?;
        }
        
        Ok(())
    }
}
```

### 5.2 XlsbReader结构

```rust
pub struct XlsbReader {
    container: XlsbContainerReader,
    sst: Option<SstTable>,
    styles: StylesRegistry,
    workbook: WorkbookReader,
    sheet_infos: Vec<SheetInfo>,
}

impl XlsbReader {
    pub fn builder() -> XlsbReaderBuilder {
        XlsbReaderBuilder::new()
    }
    
    fn new(path: &Path) -> Result<Self> {
        let container = XlsbContainerReader::open(path)?;
        
        let workbook_data = container.get_workbook_data()?;
        let workbook = WorkbookReader::deserialize(&workbook_data)?;
        
        let styles_data = container.read_entry("xl/styles.bin")?;
        let styles = StylesRegistry::deserialize(&styles_data)?;
        
        let sst = if container.has_entry("xl/sharedStrings.bin") {
            let sst_data = container.read_entry("xl/sharedStrings.bin")?;
            Some(SstTable::deserialize(&sst_data)?)
        } else {
            None
        };
        
        let sheet_infos = workbook.get_sheet_infos();
        
        Ok(Self {
            container,
            sst,
            styles,
            workbook,
            sheet_infos,
        })
    }
    
    pub fn get_sheet_infos(&self) -> &[SheetInfo] {
        &self.sheet_infos
    }
}
```

### 5.3 流式读取

```rust
impl XlsbReader {
    pub fn for_each_row(&mut self, sheet_idx: usize,
                        handler: impl RowHandler) -> Result<()> {
        if sheet_idx >= self.sheet_infos.len() {
            return Err(XlsbError::InvalidSheetIndex(sheet_idx));
        }
        
        let sheet_data = self.container.get_sheet_data(sheet_idx)?;
        let sheet_reader = SheetReader::new(sheet_data, self.sst.as_ref())?;
        sheet_reader.for_each_row(handler)?;
        Ok(())
    }
}
```

## 六、TemplateFiller API

### 6.1 Builder

```rust
pub struct TemplateFillerBuilder {
    template: Option<PathBuf>,
    output: Option<PathBuf>,
}

impl TemplateFillerBuilder {
    pub fn new() -> Self {
        Self {
            template: None,
            output: None,
        }
    }
    
    pub fn template(mut self, t: impl Into<PathBuf>) -> Self {
        self.template = Some(t.into());
        self
    }
    
    pub fn output(mut self, o: impl Into<PathBuf>) -> Self {
        self.output = Some(o.into());
        self
    }
    
    pub fn build(self) -> Result<TemplateFiller> {
        let template = self.template.ok_or(XlsbError::TemplateReadFailed("template path required"))?;
        let output = self.output.ok_or(XlsbError::TemplateReadFailed("output path required"))?;
        TemplateFiller::new(&template, &output)
    }
}
```

### 6.2 TemplateFiller结构

```rust
pub struct TemplateFiller {
    template_reader: XlsbReader,
    container_writer: XlsbContainerWriter,
    sst: SstTable,
    styles: StylesRegistry,
    template_sheets: Vec<TemplateSheet>,
    current_fill: Option<FillContext>,
}

struct TemplateSheet {
    index: usize,
    name: String,
    original_data: Bytes,
    fill_ranges: Vec<FillRange>,
}

struct FillRange {
    start_row: usize,
    start_col: usize,
    row_count: usize,
    col_count: usize,
}

struct FillContext {
    sheet_idx: usize,
    start_row: usize,
    start_col: usize,
    current_row: usize,
}
```

### 6.3 批量填充

```rust
impl TemplateFiller {
    pub fn fill_batch(&mut self, sheet_idx: usize,
                      supplier: impl CellSupplier,
                      start_row: usize, start_col: usize,
                      row_count: usize, col_count: usize) -> Result<()> {
        if sheet_idx >= self.template_sheets.len() {
            return Err(XlsbError::InvalidSheetIndex(sheet_idx));
        }
        
        let template_sheet = &self.template_sheets[sheet_idx];
        let sheet_writer = SheetWriter::from_template(&template_sheet.original_data)?;
        
        sheet_writer.fill_range(supplier, start_row, start_col, row_count, col_count)?;
        
        self.template_sheets[sheet_idx].fill_ranges.push(FillRange {
            start_row, start_col, row_count, col_count,
        });
        Ok(())
    }
}
```

### 6.4 标记填充

```rust
impl TemplateFiller {
    pub fn fill_at_marker(&mut self, sheet_idx: usize, marker: &str,
                          supplier: impl CellSupplier,
                          row_count: usize, col_count: usize) -> Result<()> {
        if sheet_idx >= self.template_sheets.len() {
            return Err(XlsbError::InvalidSheetIndex(sheet_idx));
        }
        
        let template_sheet = &self.template_sheets[sheet_idx];
        let (row, col) = template_sheet.find_marker(marker)?
            .ok_or_else(|| XlsbError::MarkerNotFound(marker.to_string()))?;
        
        self.fill_batch(sheet_idx, supplier, row, col, row_count, col_count)
    }
}
```

### 6.5 流式填充

```rust
impl TemplateFiller {
    pub fn start_fill(&mut self, sheet_idx: usize,
                      start_row: usize, start_col: usize) -> Result<()> {
        if self.current_fill.is_some() {
            return Err(XlsbError::SheetNotStarted);
        }
        
        self.current_fill = Some(FillContext {
            sheet_idx,
            start_row,
            start_col,
            current_row: start_row,
        });
        Ok(())
    }
    
    pub fn fill_rows(&mut self, cells: &[CellData]) -> Result<()> {
        let ctx = self.current_fill.as_mut()
            .ok_or(XlsbError::SheetNotStarted)?;
        
        ctx.current_row += 1;
        Ok(())
    }
    
    pub fn end_fill(&mut self) -> Result<()> {
        self.current_fill.take();
        Ok(())
    }
    
    pub fn save(&mut self) -> Result<()> {
        self.serialize()?;
        self.container_writer.finish()?;
        Ok(())
    }
}
```

## 七、使用示例

### 7.1 写入示例

```rust
use rxlsb::{XlsbWriter, CellData};

let mut writer = XlsbWriter::builder()
    .path("output.xlsb")
    .build()?;

writer.write_batch("Sheet1", |row, col| {
    match col % 4 {
        0 => CellData::text(format!("Product-{}", row)),
        1 => CellData::number(row as f64 * 100.5),
        2 => CellData::date(chrono::Utc::now()),
        3 => CellData::bool(row % 2 == 0),
        _ => CellData::blank(),
    }
}, 100_000, 4)?;

writer.close()?;
```

### 7.2 流式写入示例

```rust
let mut writer = XlsbWriter::builder()
    .path("large_output.xlsb")

writer.start_sheet("Data", 4)?;

for batch in 0..10 {
    writer.write_rows(batch * 10_000, |row, col| {
        CellData::number(row as f64 * col as f64)
    }, 10_000)?;
}

writer.end_sheet()?;
writer.close()?;
```
